   ```python
   async def _vehicle_alerts_task():
       """Daily 08:00 UTC: create Tasks for vehicle expiry alerts (D-17)."""
       import logging
       from sqlalchemy import select, and_, or_
       from datetime import timedelta
       
       log = logging.getLogger(__name__)
       SYSTEM_USER_ID = 1  # superadmin
       
       while True:
           try:
               now = datetime.now(timezone.utc)
               next_run = now.replace(hour=8, minute=0, second=0, microsecond=0)
               if now >= next_run:
                   next_run += timedelta(days=1)
               sleep_s = max(60, int((next_run - now).total_seconds()))
               await asyncio.sleep(sleep_s)
               
               async with async_session() as db:
                   from .models.vehicle import Vehicle
                   from .models.user import User
                   from .models.task import Task
                   from .models.trip import Trip
                   
                   today = datetime.now(timezone.utc).date()
                   threshold = today + timedelta(days=30)
                   
                   # 1. OSAGO expiry
                   osago_vehicles = (await db.execute(
                       select(Vehicle).where(
                           Vehicle.insurance_until.isnot(None),
                           Vehicle.insurance_until <= threshold,
                           Vehicle.insurance_until > today,
                       )
                   )).scalars().all()
                   for v in osago_vehicles:
                       tag = f"[VEHICLE:{v.id}:OSAGO_EXPIRY]"
                       existing = (await db.execute(
                           select(Task).where(Task.system_tag == tag, Task.status.in_(['todo','in_progress','planned']))
                       )).scalar_one_or_none()
                       if not existing:
                           db.add(Task(
                               title=f"Продлить ОСАГО {v.plate}",
                               description=f"ОСАГО на {v.brand} {v.model} ({v.plate}) истекает {v.insurance_until.isoformat()}",
                               category='Автотранспорт',
                               system_tag=tag,
                               due_date=v.insurance_until,
                               status='todo',
                               created_by_id=SYSTEM_USER_ID,
                               org_id=v.owner_org_id,
                           ))
                   
                   # 2. TO_WARNING — (next_to_km - current_odometer_km) < 1000
                   to_vehicles = (await db.execute(
                       select(Vehicle).where(
                           Vehicle.next_to_km.isnot(None),
                           Vehicle.current_odometer_km.isnot(None),
                           (Vehicle.next_to_km - Vehicle.current_odometer_km) < 1000,
                           (Vehicle.next_to_km - Vehicle.current_odometer_km) > -10000,  # не создавать для запущенных >10000 км просрочек
                       )
                   )).scalars().all()
                   for v in to_vehicles:
                       tag = f"[VEHICLE:{v.id}:TO_WARNING]"
                       existing = (await db.execute(
                           select(Task).where(Task.system_tag == tag, Task.status.in_(['todo','in_progress','planned']))
                       )).scalar_one_or_none()
                       if not existing:
                           km_left = v.next_to_km - v.current_odometer_km
                           db.add(Task(
                               title=f"Пройти ТО {v.plate}",
                               description=f"ТО для {v.brand} {v.model} ({v.plate}) через {km_left} км (next_to_km={v.next_to_km}, current={v.current_odometer_km})",
                               category='Автотранспорт',
                               system_tag=tag,
                               due_date=today + timedelta(days=14),
                               status='todo',
                               created_by_id=SYSTEM_USER_ID,
                               org_id=v.owner_org_id,
                           ))
                   
                   # 3. LICENSE_EXPIRY for drivers actively used in trips (last 90 days)
                   #    JOIN trips.driver_user_id → users.license_expires_at
                   recent_driver_user_ids_sub = select(Trip.driver_user_id).where(
                       Trip.driver_user_id.isnot(None),
                       Trip.date > today - timedelta(days=90),
                   ).distinct()
                   expiring_drivers = (await db.execute(
                       select(User).where(
                           User.id.in_(recent_driver_user_ids_sub),
                           User.license_expires_at.isnot(None),
                           User.license_expires_at <= threshold,
                           User.license_expires_at > today,
                       )
                   )).scalars().all()
                   for u in expiring_drivers:
                       tag = f"[DRIVER:{u.id}:LICENSE_EXPIRY]"
                       existing = (await db.execute(
                           select(Task).where(Task.system_tag == tag, Task.status.in_(['todo','in_progress','planned']))
                       )).scalar_one_or_none()
                       if not existing:
                           db.add(Task(
                               title=f"Обновить ВУ — {u.full_name}",
                               description=f"ВУ водителя {u.full_name} ({u.license_series} {u.license_number}) истекает {u.license_expires_at.isoformat()}",
                               category='Автотранспорт',
                               system_tag=tag,
                               due_date=u.license_expires_at,
                               status='todo',
                               created_by_id=SYSTEM_USER_ID,
                               assigned_user_id=u.id,  # сам водитель
                           ))
                   
                   # 4. MED_CERT_EXPIRY
                   expiring_med = (await db.execute(
                       select(User).where(
                           User.id.in_(recent_driver_user_ids_sub),
                           User.medical_cert_expires_at.isnot(None),
                           User.medical_cert_expires_at <= threshold,
                           User.medical_cert_expires_at > today,
                       )
                   )).scalars().all()
                   for u in expiring_med:
                       tag = f"[DRIVER:{u.id}:MED_CERT_EXPIRY]"
                       existing = (await db.execute(
                           select(Task).where(Task.system_tag == tag, Task.status.in_(['todo','in_progress','planned']))
                       )).scalar_one_or_none()
                       if not existing:
                           db.add(Task(
                               title=f"Обновить медсправку — {u.full_name}",
                               description=f"Медсправка водителя {u.full_name} истекает {u.medical_cert_expires_at.isoformat()}",
                               category='Автотранспорт',
                               system_tag=tag,
                               due_date=u.medical_cert_expires_at,
                               status='todo',
                               created_by_id=SYSTEM_USER_ID,
                               assigned_user_id=u.id,
                           ))
                   
                   await db.commit()
                   log.info(f"Phase 29 vehicle alerts: created Tasks for {len(osago_vehicles)} osago + {len(to_vehicles)} TO + {len(expiring_drivers)} license + {len(expiring_med)} med")
           except asyncio.CancelledError:
               raise
           except Exception as e:
               log.error(f"Phase 29 vehicle alerts task error: {e}", exc_info=True)
               await asyncio.sleep(300)  # retry in 5 min
   ```
