3. **FieldHistoryPopover.vue:**
   ```vue
   <template>
     <v-menu :close-on-content-click="false" location="bottom">
       <template #activator="{props}">
         <v-icon v-bind="props" size="16" color="grey" @click="loadHistory">mdi-information-outline</v-icon>
       </template>
       <v-card min-width="400" max-width="600">
         <v-card-title class="text-subtitle-2">История изменений — {{ fieldLabel }}</v-card-title>
         <v-divider/>
         <v-card-text v-if="loading" class="text-center pa-4"><v-progress-circular indeterminate size="24"/></v-card-text>
         <v-card-text v-else-if="!history.length" class="text-medium-emphasis">История пуста</v-card-text>
         <v-timeline v-else density="compact" side="end" class="pa-3">
           <v-timeline-item v-for="h in history" :key="h.id" dot-color="primary" size="small">
             <div class="text-caption text-medium-emphasis">{{ formatDate(h.changed_at) }} · {{ h.changed_by_name || 'system' }}</div>
             <div><span class="text-decoration-line-through text-medium-emphasis">{{ h.old_value || '—' }}</span> → <strong>{{ h.new_value || '—' }}</strong></div>
             <div v-if="h.comment" class="text-caption text-medium-emphasis">«{{ h.comment }}»</div>
           </v-timeline-item>
         </v-timeline>
       </v-card>
     </v-menu>
   </template>
   <script setup lang="ts">
   import { ref } from 'vue'
   import { apiFetch } from '@/api'
   const props = defineProps<{ vehicleId: number; fieldKey: string; fieldLabel: string }>()
   const history = ref<any[]>([])
   const loading = ref(false)
   async function loadHistory() {
     loading.value = true
     try {
       const r = await apiFetch(`/api/vehicles/${props.vehicleId}/field-history?field_key=${props.fieldKey}`)
       history.value = r.items || r
     } catch (e) { console.error(e) }
     finally { loading.value = false }
   }
   function formatDate(s: string) {
     return new Date(s).toLocaleString('ru-RU', { day:'2-digit', month:'2-digit', year:'numeric', hour:'2-digit', minute:'2-digit' })
   }
   </script>
   ```

## Risks / Gotchas
- **9 tabs structure ДОЛЖНА быть готова** (даже с stubs) — иначе нельзя протестить навигацию. Stubs внутри tabs window — допустимы.
- **`current_odometer_km` readonly** — отображается, но не редактируется (изменяется через 29-07 odometer endpoint).
- **plate editable** — пользователь может перерегистрировать ТС. Backend проверит UNIQUE при PATCH (вернёт 409 с DUPLICATE_PLATE) — frontend должен это показать.
- **JSONB props edit:** изменение текстовых полей внутри props должно отправляться backend'у как `props: {...full props object...}` или как partial `{props: {tires_type: 'новая'}}`. Backend `_coerce_patch_value` + `flag_modified` правильно обработают, если **отправляется весь props объект** при изменении.
- **Lesson 2026-05-13 flag_modified:** backend (план 29-04) уже это делает.
- **PATCH endpoint** для `_history_comment` — это служебное поле, backend (план 29-04) забирает его из body и кладёт в `VehicleFieldHistory.comment`, НЕ применяет к Vehicle.
- **Lesson 2026-05-05:** дубли let/const в большом файле — pre-push grep ОБЯЗАТЕЛЬНО.
