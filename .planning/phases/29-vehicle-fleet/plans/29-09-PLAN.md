8. **`GET /api/drivers/available?org_id={id}`** — UNION endpoint:
   ```python
   @router.get("/available")  # in external_drivers.py or trips.py
   async def list_available_drivers(org_id: int | None = None, q: str | None = None, current_user=Depends(...), db=Depends(...)):
       # 1. Internal: User WHERE can_drive=true AND (org_id IN user_organizations OR no filter)
       # 2. External: ExternalDriver WHERE (org_id=$org_id OR org_id IS NULL OR no filter)
       internal_q = select(User).where(User.can_drive == True)
       if q: internal_q = internal_q.where(User.full_name.ilike(f'%{q}%'))
       internal_users = (await db.execute(internal_q)).scalars().all()
       external_q = select(ExternalDriver)
       if org_id is not None:
           external_q = external_q.where(or_(ExternalDriver.org_id == org_id, ExternalDriver.org_id.is_(None)))
       if q: external_q = external_q.where(ExternalDriver.full_name.ilike(f'%{q}%'))
       external_drivers = (await db.execute(external_q)).scalars().all()
       return {
           "items": [
               *[{"kind":"user","id":u.id,"label":u.full_name,"license_series":u.license_series,"license_number":u.license_number,"license_expires_at":u.license_expires_at} for u in internal_users],
               *[{"kind":"external","id":d.id,"label":f"внешний: {d.full_name}","license_series":d.license_series,"license_number":d.license_number,"license_expires_at":d.license_expires_at} for d in external_drivers],
           ]
       }
   ```
   Расположить либо в `external_drivers.py` либо в `trips.py` (новый prefix `/api/drivers`). Чтобы избежать конфликта с `external_drivers.router` префиксом, использовать отдельный mini-router `drivers_router = APIRouter(prefix="/api/drivers", tags=["vehicles"])` в `external_drivers.py`, экспортировать оба роутера.
