11. **`POST /api/trips/{trip_id}/render`** — генерация .docx:
    ```python
    @router.post("/{trip_id}/render")
    async def render_trip(trip_id: int, current_user=Depends(get_current_user), db=Depends(get_db)):
        await require_action(current_user, 'vehicle.trip.create', db)
        trip = (await db.execute(select(Trip).options(selectinload(Trip.vehicle)).where(Trip.id==trip_id))).scalar_one_or_none()
        if not trip: raise HTTPException(404, detail={'code':'NOT_FOUND','message':'Путевой лист не найден'})
        # visibility check через trip.vehicle
        vehicle = trip.vehicle
        driver = await db.get(User, trip.driver_user_id) if trip.driver_user_id else await db.get(ExternalDriver, trip.driver_external_id)
        org = await db.get(Organization, vehicle.owner_org_id)
        template_path = _get_trip_template_path(vehicle.type)
        if not os.path.exists(template_path):
            raise HTTPException(500, detail={'code':'TEMPLATE_MISSING','message':f'Шаблон путёвки не найден: {os.path.basename(template_path)}'})
        try:
            tpl = DocxTemplate(template_path)
            ctx = _build_trip_context(trip, vehicle, driver, org)
            tpl.render(ctx)
            buf = BytesIO()
            tpl.save(buf)
            buf.seek(0)
        except Exception as e:
            import traceback
            raise HTTPException(500, detail={'code':'TEMPLATE_RENDER_ERROR','message':'Ошибка генерации путёвки','error_class':e.__class__.__name__,'error_raw':str(e),'traceback':traceback.format_exc()[-1500:],'hint':'Проверьте Lessons 2026-05-15: {% tr %} запрещён'})
        # mark trip.status='rendered'
        trip.status = 'rendered'
        await db.commit()
        filename = f"trip_{trip_id}_{trip.date.isoformat()}.docx"
        return StreamingResponse(buf, media_type="application/vnd.openxmlformats-officedocument.wordprocessingml.document", headers={"Content-Disposition": f'attachment; filename="{filename}"'})
    ```
