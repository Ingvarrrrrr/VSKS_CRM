3. **Helpers:**

   ```python
   def _bool_from_cell(val):
       if val is None: return None
       s = str(val).strip().lower()
       if s in ('да','есть','исправно','норм','+','true','yes','1'): return True
       if s in ('нет','отсутствует','-','false','no','0',''): return False
       return None
   
   def _date_from_cell(val):
       if val is None: return None
       if isinstance(val, (datetime, date)): return val if isinstance(val, date) and not isinstance(val, datetime) else val.date()
       try: return date.fromisoformat(str(val).strip()[:10])
       except: return None
   
   def _map_type(val):
       if val is None: return 'other'
       s = str(val).strip().lower()
       for k, v in TYPE_MAP.items():
           if k in s: return v
       return 'other'
   
   def _map_state(val):
       if val is None: return 'working'
       s = str(val).strip().lower()
       for k, v in STATE_MAP.items():
           if k in s: return v
       return 'working'
   
   def _split_brand_model(val):
       """'Митсубиши Делика' -> ('Митсубиши', 'Делика'). Если одно слово — brand=val, model=''."""
       if not val: return ('', '')
       parts = str(val).strip().split(None, 1)
       return (parts[0], parts[1] if len(parts) > 1 else '')
   ```
4. **Owner lookup:**
   ```python
   async def _lookup_owner_org_id(db, raw_owner: str) -> tuple[int | None, str]:
       """ВСКС/ВСС/имя организации → org_id. Иначе → None, raw_owner идёт в assigned_text."""
       if not raw_owner: return (None, '')
       s = str(raw_owner).strip()
       if not s: return (None, '')
       # Try exact match by name
       org = (await db.execute(select(Organization).where(Organization.name.ilike(s)))).scalar_one_or_none()
       if org: return (org.id, '')
       # Try substring "ВСКС"
       if 'вскс' in s.lower():
           org = (await db.execute(select(Organization).where(Organization.name.ilike('%ВСКС%')))).scalar_one_or_none()
           if org: return (org.id, '')
       return (None, s)  # личное имя или регион → assigned_text
   ```
5. **Main function:**
   ```python
   async def seed_vehicles_from_xlsx(db: AsyncSession, xlsx_path: str | None = None) -> dict:
       """Idempotent seed. ON CONFLICT (plate) DO NOTHING.
       Returns {inserted: N, skipped: M, unmapped_regions: [...]}."""
       if xlsx_path is None:
           xlsx_path = os.path.join(os.path.dirname(__file__), '..', '..', '..', 'Доработки', 'реестр_транспорта_от_Голичкова_обновление_042026.xlsx')
       if not os.path.exists(xlsx_path):
           log.warning(f"vehicles_seed: xlsx not found at {xlsx_path}, skipping")
           return {"inserted": 0, "skipped": 0, "unmapped_regions": [], "reason": "xlsx_not_found"}
       
       # DEFAULT owner: ВСКС (id=1, INN 7731178803 по Gotchas)
       default_owner_org_id = 1
       
       wb = openpyxl.load_workbook(xlsx_path, read_only=True, data_only=True)
       if 'Лист2' not in wb.sheetnames:
           log.warning(f"vehicles_seed: sheet 'Лист2' not found")
           return {"inserted": 0, "skipped": 0, "unmapped_regions": [], "reason": "no_sheet"}
       ws = wb['Лист2']
       rows = list(ws.iter_rows(values_only=True))[1:]  # skip header
       
       inserted = 0
       skipped = 0
       unmapped_regions = set()
       
       for row in rows:
           plate = row[4]
           if not plate: continue
           plate = str(plate).strip().upper()
           
           vin = row[3]
           if vin and str(vin).strip().lower() not in ('отсутствует', 'нет', ''):
               vin = str(vin).strip().upper()[:17]
           else:
               vin = None
           
           brand_model_raw = row[1] or ''
           brand, model = _split_brand_model(brand_model_raw)
           
           color = (row[2] or '')[:50] if row[2] else None
           registered_at = _date_from_cell(row[6])
           
           # Owner lookup
           owner_org_id, owner_fallback_text = await _lookup_owner_org_id(db, row[5])
           if owner_org_id is None:
               owner_org_id = default_owner_org_id  # ВСКС fallback
               # owner_fallback_text → лежит в notes если хочется
           
           # Assigned text (column 7 "У кого в эксплуатации" — обычно регион/имя)
           assigned_text_raw = (str(row[7]).strip() if row[7] else '')[:100]
           if assigned_text_raw:
               unmapped_regions.add(assigned_text_raw)  # для UI mapping в плане 29-12
           
           v_type = _map_type(row[8])
           v_state = _map_state(row[10])
           insurance_until = _date_from_cell(row[9])
           
           props = {
               'tires_type': str(row[15] or '')[:200],
               'branding': str(row[16] or '')[:200],
               'paint_condition': str(row[21] or '')[:200],
               'defect_description': str(row[22] or '')[:500],
               'note': str(row[23] or '')[:500],
           }
           
           result = await db.execute(text("""
               INSERT INTO vehicles (
                   owner_org_id, assigned_org_id, assigned_text,
                   brand, model, color, vin, plate, registered_at,
                   type, state, insurance_until,
                   has_tracker, akb_ok, has_radio, mirrors_ok,
                   has_keys, has_first_aid_kit, has_spare_wheel, has_extinguisher,
                   props, created_at, updated_at
               ) VALUES (
                   :owner_org_id, NULL, :assigned_text,
                   :brand, :model, :color, :vin, :plate, :registered_at,
                   :type, :state, :insurance_until,
                   :has_tracker, :akb_ok, :has_radio, :mirrors_ok,
                   :has_keys, :has_first_aid_kit, :has_spare_wheel, :has_extinguisher,
                   :props::jsonb, NOW(), NOW()
               )
               ON CONFLICT (plate) DO NOTHING
           """), {
               "owner_org_id": owner_org_id, "assigned_text": assigned_text_raw,
               "brand": brand[:100], "model": model[:100], "color": color,
               "vin": vin, "plate": plate[:20], "registered_at": registered_at,
               "type": v_type, "state": v_state, "insurance_until": insurance_until,
               "has_tracker": _bool_from_cell(row[11]), "akb_ok": _bool_from_cell(row[12]),
               "has_radio": _bool_from_cell(row[13]), "mirrors_ok": _bool_from_cell(row[14]),
               "has_keys": _bool_from_cell(row[17]), "has_first_aid_kit": _bool_from_cell(row[18]),
               "has_spare_wheel": _bool_from_cell(row[19]), "has_extinguisher": _bool_from_cell(row[20]),
               "props": __import__('json').dumps(props, ensure_ascii=False),
           })
           if result.rowcount > 0:
               inserted += 1
           else:
               skipped += 1
       
       await db.commit()
       log.info(f"vehicles_seed: inserted={inserted}, skipped={skipped}, unmapped_regions={len(unmapped_regions)}")
       return {"inserted": inserted, "skipped": skipped, "unmapped_regions": sorted(unmapped_regions)}
   ```

## Risks / Gotchas
- **xlsx path:** в проде путь относительно WORKDIR контейнера. Если файл не копируется в Docker image — seed безопасно skip'нется. **TODO для DevOps:** добавить файл в Dockerfile / mount volume. На время этой фазы — если не найден, не падать.
- **ON CONFLICT(plate):** UNIQUE constraint на plate ОБЯЗАТЕЛЬНО присутствует (план 29-02).
- **Encoding:** Cyrillic xlsx headers — openpyxl читает корректно (R-5 verified).
- **brand_model split** — простой split по первому пробелу. Для «КАМАЗ-43118» split не подходит, brand='КАМАЗ-43118', model=''. OK для seed; пользователь поправит вручную через UI.
- **owner_org_id fallback на ВСКС id=1** — закодировано хардкодом, по Gotcha 2026-04-23 это Organization id=1 (ИНН 7731178803). Если в проде org_id для ВСКС другой — seed создаст машины с неверным owner.
