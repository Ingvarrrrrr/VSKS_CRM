    import { test, expect } from '@playwright/test'

    test.describe('Phase 15: PurchaseItemsEditor parity — Заявка ↔ Новый заказ', () => {
      test.beforeEach(async ({ page }) => {
        await login(page, 'admin', 'admin123')
        await waitForOverlays(page)
      })

      test('ITEMS-EDITOR-02: CreateOrderView — add row, pick product from catalogue, photo tooltip visible', async ({ page }) => {
        const apiErrors = collectApiErrors(page)
        await page.goto('/create-order')
        await waitForOverlays(page)
        // click "+ Добавить позицию"
        await page.getByRole('button', { name: /Добавить позицию/i }).first().click()
        // click the name cell → product picker opens
        await page.locator('textarea[placeholder*="Наименование"], [data-test="item-name-picker"]').first().click()
        // search for a known product
        await page.getByPlaceholder(/Поиск товара/i).fill('Тестовый')
        // select first result
        await page.locator('.v-list-item').first().click()
        // photo thumb appears in row — hover it → tooltip
        const thumb = page.locator('.purchase-items-editor .v-table tbody tr').first().locator('img').first()
        await expect(thumb).toBeVisible({ timeout: 5000 })
        expect(apiErrors).toEqual([])
      })

      test('ITEMS-EDITOR-03: CreateOrderView — full product card dialog with photo upload', async ({ page }) => {
        const apiErrors = collectApiErrors(page)
        await page.goto('/create-order')
        await waitForOverlays(page)
        await page.getByRole('button', { name: /Добавить позицию/i }).first().click()
        await page.locator('textarea[placeholder*="Наименование"]').first().click()

        await page.getByLabel(/Наименование/i).fill(`E2E товар ${Date.now()}`)
        await page.getByLabel(/Категория/i).fill('Тест')
        await page.getByLabel(/Цена за единицу/i).fill('100')
        // upload photo
        const fileInput = page.locator('input[type="file"]').first()
        await fileInput.setInputFiles(SAMPLE_PHOTO)
        await expect(page.locator('img[src^="blob:"]')).toBeVisible({ timeout: 3000 })
        await page.getByRole('button', { name: /Сохранить/i }).click()
        // back in the row — photo thumb now visible
        await expect(page.locator('.purchase-items-editor .v-table tbody tr').first().locator('img').first()).toBeVisible({ timeout: 5000 })
        expect(apiErrors).toEqual([])
      })

      test('ITEMS-EDITOR-04: CreateOrderView — Excel import 2-step drag-and-drop mapping', async ({ page }) => {
        const apiErrors = collectApiErrors(page)
        await page.goto('/create-order')
        await waitForOverlays(page)
        await page.getByRole('button', { name: /Импорт из файла/i }).click()
        // Step 1: upload

        await expect(page.locator('.purchase-items-editor .v-table tbody tr')).toHaveCount(3, { timeout: 5000 })
        expect(apiErrors).toEqual([])
      })

      test('ITEMS-EDITOR-06: CreateOrderView — row auto-calc + save payload strips helpers', async ({ page }) => {
        const apiErrors = collectApiErrors(page)

        await page.goto('/create-order')
        await waitForOverlays(page)
        // minimal valid form (subsidy, FEO 3 levels, 1 item) — adapt selectors to real form

        // auto-calc — verify total_price rendered = 1000
        await expect(page.locator('.purchase-items-editor tbody tr').first()).toContainText('1 000')

        await page.getByRole('button', { name: /Сохранить/i }).click()

        const body = req.postDataJSON()
        expect(body.items).toBeDefined()

        })
        await page.getByRole('button', { name: /Импорт из файла/i }).click()
        await page.locator('input[type="file"]').last().setInputFiles(SAMPLE_XLSX)
        await page.getByRole('button', { name: /Далее|Загрузить/i }).click()
        await page.getByRole('button', { name: /Применить/i }).click()

        await expect(page.locator('.purchase-items-editor tbody tr')).toHaveCount(3, { timeout: 5000 })
        expect(apiErrors).toEqual([])
      })
