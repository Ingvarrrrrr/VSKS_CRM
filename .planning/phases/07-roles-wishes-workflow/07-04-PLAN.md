# Check if new assigned user is a subordinate of current_user
if new_assigned_user_id and new_assigned_user_id != purchase.assigned_user_id:
    sub_res = await db.execute(

            UserHierarchy.manager_id == current_user.id
        )

    subordinate_ids = [r[0] for r in sub_res.all()]
    is_subordinate = new_assigned_user_id in subordinate_ids

    if not is_subordinate:
        # D-15: Non-subordinate requires consent flow
        # Check if PurchaseMember row exists for this user, create if not
        from app.models.purchase_event import PurchaseMember
        existing = await db.execute(
            select(PurchaseMember).where(
                PurchaseMember.purchase_id == purchase.id,
                PurchaseMember.user_id == new_assigned_user_id,
            )
        )
        member = existing.scalar_one_or_none()
        if not member:
            member = PurchaseMember(
                purchase_id=purchase.id,
                user_id=new_assigned_user_id,
                role="executor",
                added_by_id=current_user.id,
                consent_pending=True,
            )
            db.add(member)
        else:
            member.consent_pending = True

        # D-16: Send consent request notification via chat
        purchase_title = purchase.subject or purchase.item_name or f"Закупка #{purchase.id}"
        await chat_manager.send_to_user(new_assigned_user_id, {
            "type": "system_notification",

**2. If there is NO consent endpoint for purchases yet**, create one:

```python
@router.post("/{purchase_id}/consent")
async def accept_purchase_consent(
    purchase_id: int,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Accept consent to be executor on a purchase."""
    from app.models.purchase_event import PurchaseMember
    member = await db.execute(
        select(PurchaseMember).where(
            PurchaseMember.purchase_id == purchase_id,
            PurchaseMember.user_id == current_user.id,
            PurchaseMember.consent_pending == True,
        )
    )
    pm = member.scalar_one_or_none()
    if not pm:
        raise HTTPException(404, "Нет ожидающего запроса согласия")
    pm.consent_pending = False
    # Set as executor
    purchase = await db.get(Purchase, purchase_id)
    purchase.assigned_user_id = current_user.id
    await db.commit()
    return {"status": "accepted", "purchase_id": purchase_id}
```
